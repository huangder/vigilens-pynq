cmake_minimum_required(VERSION 3.16)
project(vigilens_mnn_probe LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MNN_SOURCE_DIR "" CACHE PATH "Official MNN 3.6.1 source root")
option(VG_MNN_CONVERTER "Build host ONNX converter" OFF)
if(NOT EXISTS "${MNN_SOURCE_DIR}/include/MNN/Interpreter.hpp")
  message(FATAL_ERROR "Set MNN_SOURCE_DIR to official MNN 3.6.1")
endif()
foreach(feature BUILD_SHARED_LIBS SEP_BUILD BUILD_TOOLS BUILD_TEST BUILD_BENCHMARK
                BUILD_TRAIN BUILD_DEMO BUILD_QUANTOOLS BUILD_PROTOBUFFER BUILD_TORCH
                BUILD_LLM BUILD_DIFFUSION INTERNAL METAL OPENCL OPENGL VULKAN CUDA
                MUSA TENSORRT COREML NNAPI QNN HEXAGON ARM82 SUPPORT_FP16_ARMV7
                SUPPORT_BF16 LOW_MEMORY KLEIDIAI SME2 OPENMP)
  set(MNN_${feature} OFF CACHE BOOL "Disabled for fixed FP32 CPU probe" FORCE)
endforeach()
set(MNN_BUILD_CONVERTER ${VG_MNN_CONVERTER} CACHE BOOL "Host converter only" FORCE)
set(MNN_USE_THREAD_POOL ON CACHE BOOL "CPU threading" FORCE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^armv7")
  set(MNN_BUILD_HARD ON CACHE BOOL "ARM hard float" FORCE)
else()
  set(MNN_BUILD_HARD OFF CACHE BOOL "Not an ARM build" FORCE)
endif()
add_subdirectory("${MNN_SOURCE_DIR}" mnn-runtime EXCLUDE_FROM_ALL)
add_executable(vg_mnn_probe model_probe.cc)
target_include_directories(vg_mnn_probe SYSTEM PRIVATE "${MNN_SOURCE_DIR}/include")
target_link_libraries(vg_mnn_probe PRIVATE MNN)
target_compile_options(vg_mnn_probe PRIVATE -Wall -Wextra -Werror)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(vg_mnn_probe PRIVATE -static-libstdc++ -static-libgcc)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^armv7")
  target_compile_options(vg_mnn_probe PRIVATE -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard)
endif()
if(VG_MNN_CONVERTER)
  add_executable(audit_mnn_model audit_model.cc)
  target_include_directories(audit_mnn_model SYSTEM PRIVATE
    "${MNN_SOURCE_DIR}/schema/current" "${MNN_SOURCE_DIR}/3rd_party/flatbuffers/include")
  target_compile_options(audit_mnn_model PRIVATE -Wall -Wextra -Werror)
endif()
