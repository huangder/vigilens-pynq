option(VG_BUILD_MODEL_PROBE "Build the real TFLite model feasibility probe" ON)
option(VG_ATTENTION_OPS "Register the original MediaPipe attention custom operators" ON)
option(VG_ENABLE_XNNPACK "Enable the CPU XNNPACK delegate for target benchmarking" OFF)
option(VG_ENABLE_RUY "Enable optimized RUY kernels for target benchmarking" OFF)
if(VG_BUILD_MODEL_PROBE)
  if(NOT EXISTS "${VG_TENSORFLOW_SOURCE}/tensorflow/lite/CMakeLists.txt")
    message(FATAL_ERROR "Set VG_TENSORFLOW_SOURCE to an unpacked TensorFlow v2.16.1 source tree")
  endif()
  set(TFLITE_ENABLE_XNNPACK ${VG_ENABLE_XNNPACK} CACHE BOOL "" FORCE)
  set(TFLITE_ENABLE_RUY ${VG_ENABLE_RUY} CACHE BOOL "" FORCE)
  set(TFLITE_ENABLE_GPU OFF CACHE BOOL "" FORCE)
  set(TFLITE_ENABLE_NNAPI OFF CACHE BOOL "" FORCE)
  set(TFLITE_ENABLE_EXTERNAL_DELEGATE OFF CACHE BOOL "" FORCE)
  set(TFLITE_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${VG_TENSORFLOW_SOURCE}/tensorflow/lite" tflite EXCLUDE_FROM_ALL)
  add_executable(vg_model_probe model_probe.cc)
  target_link_libraries(vg_model_probe PRIVATE tensorflow-lite)
  if(VG_ATTENTION_OPS)
    include(mediapipe_ops.cmake)
    target_link_libraries(vg_model_probe PRIVATE vg_mediapipe_ops)
    target_compile_definitions(vg_model_probe PRIVATE VG_ATTENTION_OPS=1)
  endif()
endif()
