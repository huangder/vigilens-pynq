cmake_minimum_required(VERSION 3.16)
project(vigilens_ncnn_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NCNN_SOURCE_DIR "" CACHE PATH "Unpacked official ncnn 20240410 source")
option(VG_NCNN_OPENMP "Enable NCNN OpenMP runtime" ON)
option(VG_NCNN_INT8 "Build an INT8-enabled candidate probe (requires calibrated model)" OFF)
if(NOT EXISTS "${NCNN_SOURCE_DIR}/src/net.h")
  message(FATAL_ERROR "Set NCNN_SOURCE_DIR to the unpacked official source root")
endif()
foreach(feature VULKAN BUILD_TOOLS BUILD_TESTS BUILD_BENCHMARK BUILD_EXAMPLES BENCHMARK
                BF16 VFPV4 ARM82 ARM82DOT ARM82FP16FML ARM84BF16 ARM84I8MM
                ARM86SVE ARM86SVE2 ARM86SVEBF16 ARM86SVEI8MM ARM86SVEF32MM)
  set(NCNN_${feature} OFF CACHE BOOL "Disabled for FP32 Cortex-A9 budget probe" FORCE)
endforeach()
set(NCNN_OPENMP ${VG_NCNN_OPENMP} CACHE BOOL "Probe OpenMP choice" FORCE)
set(NCNN_INT8 ${VG_NCNN_INT8} CACHE BOOL "Probe INT8 choice" FORCE)
set(NCNN_SHARED_LIB OFF CACHE BOOL "Static ncnn" FORCE)
set(NCNN_STDIO ON CACHE BOOL "Load local parameter/model files" FORCE)
set(NCNN_STRING ON CACHE BOOL "Named input/output blobs" FORCE)
add_subdirectory("${NCNN_SOURCE_DIR}" ncnn-runtime EXCLUDE_FROM_ALL)
add_executable(vg_ncnn_probe model_probe.cc)
target_compile_definitions(vg_ncnn_probe PRIVATE
  VG_PROBE_INT8_ENABLED=$<BOOL:${VG_NCNN_INT8}>
  VG_PROBE_OPENMP_ENABLED=$<BOOL:${VG_NCNN_OPENMP}>)
target_link_libraries(vg_ncnn_probe PRIVATE ncnn)
target_compile_options(vg_ncnn_probe PRIVATE -Wall -Wextra -Werror)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|armv7|armv7l|armv7-a)$")
  # NCNN 20240410 uses compiler __ARM_NEON, not an NCNN_ARM_NEON option.
  target_compile_options(ncnn PRIVATE -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard)
  target_compile_options(vg_ncnn_probe PRIVATE -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(vg_ncnn_probe PRIVATE -static-libstdc++ -static-libgcc)
endif()
